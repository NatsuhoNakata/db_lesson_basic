--Q6　テーブル・レコード・カラムという3つの単語を適切に使用して、下記のSQL文を日本語で説明。

SELECT
  `name`, `email`, `age`
FROM
  `people`
WHERE
  `department_id` = 1
ORDER BY
  `created_at`;

--説明:peopleテーブルから、departments_id=1に所属する人のname, email,ageのカラムを指定し、作成日時の昇順で取得した状態のレコードを表示させる。
